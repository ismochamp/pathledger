cmake_minimum_required(VERSION 3.16)
project(PathLedger LANGUAGES CXX)
add_executable(pathledger src/pathledger.cpp)
target_compile_features(pathledger PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(pathledger PRIVATE /W4 /utf-8)
else()
  target_compile_options(pathledger PRIVATE -Wall -Wextra -Wpedantic)
endif()
